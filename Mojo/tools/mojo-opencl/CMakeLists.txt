cmake_minimum_required(VERSION 3.22)
project(MojoOpenCLCompiler LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
if(LLVM_VERSION_MAJOR LESS 22)
  message(FATAL_ERROR "The standalone emitter requires LLVM 22 or newer")
endif()
get_filename_component(MODULAR_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(EMITTER_DIR "${MODULAR_ROOT}/Mojo/lib/Compiler/ObjectCompiler/Target/OpenCL")
add_executable(mojo-opencl-emit emit.cpp "${EMITTER_DIR}/OpenCLEmitter.cpp")
target_compile_features(mojo-opencl-emit PRIVATE cxx_std_17)
target_include_directories(mojo-opencl-emit SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS})
target_include_directories(mojo-opencl-emit PRIVATE "${EMITTER_DIR}" "${MODULAR_ROOT}/Target/include")
target_compile_options(mojo-opencl-emit PRIVATE -Wall -Wextra -Werror)
llvm_map_components_to_libnames(LLVM_LIBRARIES Core IRReader Support Passes TransformUtils ScalarOpts InstCombine Analysis)
target_link_libraries(mojo-opencl-emit PRIVATE ${LLVM_LIBRARIES})
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_program(OPENCL_CLANG NAMES clang HINTS "${LLVM_TOOLS_BINARY_DIR}" REQUIRED)
enable_testing()
add_test(NAME emitter_rejections
  COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/test_emitter.py"
    "$<TARGET_FILE:mojo-opencl-emit>" "${OPENCL_CLANG}")
